function initials(n){return n.split(" ").map(x=>x[0]).slice(0,2).join("").toUpperCase()}
function scholarMap(){return Object.fromEntries(scholars.map(s=>[s.id,s]))}
function badge(status){
 const cls=/Completed|Published|Accepted|Submitted/.test(status)?"ok":/Pending|Upcoming|Draft|Review|Revision/.test(status)?"warn":/Rejected|Overdue/.test(status)?"danger":"";
 return `<span class="badge ${cls}">${status}</span>`;
}
function applyFilters(){
 const q=$("globalSearch").value.toLowerCase();
 const g=$("guideFilter").value, y=$("yearFilter").value;
 filteredScholars=scholars.filter(s=>(!q||[s.name,s.id,s.topic,s.dept].join(" ").toLowerCase().includes(q))&&(!g||s.guide===g)&&(!y||String(s.year)===y));
 renderAll();
}
function renderKpis(){

 ["Total Scholars",scholars.length],["Active Scholars",scholars.length],["Average Progress",avg+"%"],["Publications",publications.length],
 ["Scopus",publications.filter(p=>p.indexing==="Scopus").length],["Q1",publications.filter(p=>p.quartile==="Q1").length],["Submitted / Review",publications.filter(p=>["Submitted","Under Review","Revision"].includes(p.status)).length],["Coursework Complete",scholars.filter(s=>s.coursework==="4/4").length+"/"+scholars.length]
 ].map(x=>`<div class="card"><div class="label">${x[0]}</div><div class="value">${x[1]}</div></div>`).join("");
}
